#include "algoritmos.h"

int compara(int a, int b, struct estatisticas *e)
{
  if (e != NULL)
    e->comparacoes++;

  if (a > b)
    return(1);
  if (a == b)
    return(0);
  return(-1);
}

void troca(int v[], int i, int j, struct estatisticas *e)
{
  if (i == j)
    return;
  int aux = v[i];
  v[i] = v[j];
  v[j] = aux;

  if (e != NULL)
    e->trocas++;
}


void particao(int v[], int esq, int dir, int *pos_pivo,
struct estatisticas *e)
{
  int i, j;
  int meio = esq + (dir - esq) / 2;

  if (compara(v[meio], v[esq], e) < 0) // aplicação do pivô como mediana entre o primeiro, meio e último elementos do (sub-)vetor
    troca(v, meio, esq, e);

  if (compara(v[dir], v[esq], e) < 0)
    troca(v, dir, esq, e);

  if (compara(v[dir], v[meio], e) < 0)
    troca(v, dir, meio, e);

  troca(v, esq, meio, e); // Coloca a mediana na primeira posição
  int pivo = v[esq];
  i = esq + 1;
  j = dir;

  while (1) // Particiona o vetor em relação ao pivô
  {
    while (i <= dir && compara(v[i], pivo, e) < 0)
      i++;

    while (j > esq && compara(v[j], pivo, e) > 0)
      j--;

    if (i >= j)
      break;

    troca(v, i, j, e);
    i++;
    j--;
  }

  troca(v, esq, j, e);
  *pos_pivo = j;
}


void quicksort(int v[], int esq, int dir, struct estatisticas *e)
{

  if (v == NULL || esq >= dir)
    return;

  int pos_pivo;

  if (esq < dir)
  {
    particao(v, esq, dir, &pos_pivo, e);
    quicksort(v, esq, pos_pivo - 1, e);
    quicksort(v, pos_pivo + 1, dir, e);
  }
}


void insertsort(int v[], int n, struct estatisticas *e)
{
  int i, j, aux;

  if (v == NULL || n <= 1)
    return;

  for (i = 1; i < n; i++)
  {
    aux = v[i];
    j = i - 1;

    while (j >= 0)
    {
      if (compara(v[j], aux, e) <= 0)
        break;

      v[j + 1] = v[j]; // desloca elementos pra direita
      if (e != NULL)
        e->trocas++;
      j--;
    }

    v[j + 1] = aux; // Insere o elemento na posição correta
  }
}


void sacodeheap_int(int m, int v[], int i, struct estatisticas *e)
{
  if (v == NULL || m <= 0 || i < 0 || i >= m)
    return;

  while (2 * i + 1 < m)
  {
    int filho = 2 * i + 1;

    if (filho + 1 < m)
    {
      if (e != NULL)
        e->comparacoes++;

      if (v[filho] < v[filho + 1])
        filho++;
    }
    if (e != NULL)
      e->comparacoes++;
    if (v[i] >= v[filho])
      break;

    troca(v, i, filho, e);
    i = filho;
  }
}


void heapsort(int vetor[], int n, struct estatisticas *e)
{
  if (vetor == NULL || n <= 1)
    return;

  for (int i = n / 2 - 1; i >= 0; i--)
    sacodeheap_int(n, vetor, i, e);

  for (int i = n - 1; i > 0; i--)
  {
    troca(vetor, 0, i, e);
    sacodeheap_int(i, vetor, 0, e);
  }
}


long sorteianum(long min, long max) //retorna elementos entre min e max
{
  long num = (min + rand() % (max - (min) + 1));
  return (num);
}
